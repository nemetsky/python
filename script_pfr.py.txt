def send_config_commands(device, config_commands):
    with netmiko.ConnectHandler(**device) as ssh:
        #ssh.send_command_timing("terminal datadump")
        #ssh.send_command_timing("configure terminal")
        output = ssh.send_config_set(config_commands) #cmd_verify=False, enter_config_mode=False)
        output += ssh.send_command_timing("commit")
        output += ssh.send_command_timing("confirm")
    return output   

def send_show_command(device, command):
    with netmiko.ConnectHandler(**device) as ssh:
        output = ssh.send_command(command)
        return output
