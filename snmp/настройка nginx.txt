[Unit]

After=network.target

[Service]
# Имя пользователя Linux, под которым будет работать скрипт (например, root или ubuntu)
User=admin-st
Group=admin-st

# Путь к папке, где лежит ваш app.py
WorkingDirectory=/opt/scripts

# Путь к исполняемому файлу gunicorn. 
# Если использовали venv, путь будет: /path/to/your/project/venv/bin/gunicorn
# Если ставили глобально, то просто: /usr/local/bin/gunicorn (можно узнать через команду `which gunicorn`)
ExecStart=/home/admin-st/.local/bin/gunicorn --workers 3 --bind 127.0.0.1:5001 script_flask_v2:app

Restart=always
RestartSec=5

[Install]
WantedBy=multi-user.target
"""
